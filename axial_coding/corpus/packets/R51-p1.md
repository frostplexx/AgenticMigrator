```javascript
async function initialize() {
  const container = document.getElementById("feed");
  if (!container) return;

  try {
    const response = await fetch("http://tecnohoy.com.ve/feed");
    const text = await response.text();
    const parser = new DOMParser();
    const xmlDoc = parser.parseFromString(text, "text/xml");

    const entries = xmlDoc.querySelectorAll("entry, item");

    let count = 1;
    const limit = Math.min(entries.length, 10);

    for (let i = 0; i < limit; i++) {
      const entry = entries[i];

      // Atom uses <title>, RSS uses <title>
      const titleNode = entry.querySelector("title");
      const title = titleNode ? titleNode.textContent : "No Title";

      // Atom uses <link href="...">, RSS uses <link> text content
      let link = "#";
      const linkNode = entry.querySelector("link");
      if (linkNode) {
        link = linkNode.getAttribute("href") || linkNode.textContent;
      }

      const div = document.createElement("div");
      div.innerHTML = `<h5>${count++}. <a href="${link}" target="_blank">${title}</a></h5>`;
      container.appendChild(div);
    }
  } catch (error) {
    console.error("Error loading feed:", error);
    container.innerHTML = "Error loading feed.";
  }
}
```
